     6	class ExponentialRetryBackoff(baseDelay: Duration, maxDelay: Duration) {
     7	  private val baseMillis: Long
     8	  private val maxMillis: Long
     9	
    10	  init {
    11	    require(validDelays(baseDelay, maxDelay)) { "재시도 지연 설정이 허용 범위를 벗어났습니다." }

    13	    maxMillis = maxDelay.toMillis()
    14	  }
    15	
    16	  /** 1부터 시작하는 재시도 번호. cap에 도달하면 반복을 끝내 큰 번호에도 overflow가 없다. */
    17	  fun upperBoundMillis(retryNumber: Int): Long {
    18	    require(retryNumber >= 1) { "재시도 번호는 1 이상이어야 합니다." }
    19	    var bound = baseMillis
    20	    var number = 1
    21	    while (number < retryNumber && bound < maxMillis) {
    22	      bound = if (bound > maxMillis / 2) maxMillis else bound * 2
    23	      number++
    24	    }
    25	    return bound
    26	  }

    22	 */
    23	class RetryOnTransientFailureInterceptor(
    24	  private val maxRetries: Int,
    25	  private val backoff: ExponentialRetryBackoff,
    26	  private val jitter: RetryJitter,
    27	  private val sleeper: RetrySleeper,
    28	  private val readTimeout: Duration? = null,
    29	) : ClientHttpRequestInterceptor {
    30	  private val log = LoggerFactory.getLogger(javaClass)
    31	
    32	  init {
    33	    require(maxRetries in 0..3) { "추가 재시도 횟수는 0부터 3까지여야 합니다." }
    34	  }
    35	
    36	  override fun intercept(
    37	    request: HttpRequest,
    38	    body: ByteArray,
    39	    execution: ClientHttpRequestExecution,
    40	  ): ClientHttpResponse {
    41	    val oldAttempt = MDC.get("attempt")
    42	    try {
    43	      var retries = 0
    44	      while (true) {
    45	        stopIfInterrupted()
    46	        request.headers.set("X-Internal-Attempt", (retries + 1).toString())
    47	        MDC.put("attempt", (retries + 1).toString())
    48	        log.info("event=remote.attempt.started attempt={}", retries + 1)
    49	        try {
    50	          val started = System.nanoTime()
    51	          val response = execution.execute(request, body)
    52	          return (readTimeout?.let { ReadDeadline.wrap(response, started, it) } ?: response).also {
    53	            log.info(
    54	              "event=remote.attempt.completed attempt={} status={}",
    55	              retries + 1,
    56	              it.statusCode.value(),
    57	            )
    58	          }
    59	        } catch (e: IOException) {
    60	          log.error(
    61	            "event=remote.failed attempt={} exception={}",
    62	            retries + 1,
    63	            e.javaClass.simpleName,
    64	          )
    65	          stopIfInterrupted()
    66	          if (retries >= maxRetries || !isRetryable(request, e)) throw e
    67	          retries++
    68	          val upper = backoff.upperBoundMillis(retries)
    69	          val delay = jitter.sample(upper)
    70	          check(delay in 0..upper) { "재시도 난수 결과가 허용 범위를 벗어났습니다." }
    71	          log.info(
    72	            "event=remote.retry.scheduled nextAttempt={} delayMs={} reason={}",
    73	            retries + 1,
    74	            delay,
    75	            e.javaClass.simpleName,
    76	          )
    77	          try {
    78	            sleeper.sleep(delay)
    79	          } catch (_: InterruptedException) {
    80	            Thread.currentThread().interrupt()
    81	            throw interrupted()
    82	          }
    83	          stopIfInterrupted()
    84	        }
    85	      }
    86	    } finally {
    87	      if (oldAttempt == null) MDC.remove("attempt") else MDC.put("attempt", oldAttempt)
    88	    }
    89	  }

    96

    98	    request.method in RETRYABLE_METHODS &&

   100	        .any {
   101	          it is ConnectException || it is HttpTimeoutException || it is SocketTimeoutException
   102	        }
