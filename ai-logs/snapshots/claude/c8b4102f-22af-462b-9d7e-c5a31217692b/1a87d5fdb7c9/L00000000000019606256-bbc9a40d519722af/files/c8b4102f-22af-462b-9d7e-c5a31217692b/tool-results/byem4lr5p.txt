    56	internal class NumericCidr(value: String) {
    57	  private val address: ByteArray
    58	  private val bits: Int
    59	
    60	  init {
    61	    val parts = value.split('/')
    62	    require(parts.size == 2) { "내부 호출 CIDR 형식이 올바르지 않습니다." }
    63	    address = numeric(parts[0]) ?: throw IllegalArgumentException("내부 호출 CIDR 주소가 올바르지 않습니다.")
    64	    bits = parts[1].toIntOrNull() ?: throw IllegalArgumentException("내부 호출 CIDR 길이가 올바르지 않습니다.")
    65	    require(bits in 0..address.size * 8) { "내부 호출 CIDR 길이가 올바르지 않습니다." }
    66	  }
    67	
    68	  fun contains(value: String): Boolean {
    69	    val remote = numeric(value) ?: return false
    70	    if (remote.size != address.size) return false
    71	    return (0 until bits).all { bit ->
    72	      val mask = 1 shl (7 - bit % 8)
    73	      (remote[bit / 8].toInt() and mask) == (address[bit / 8].toInt() and mask)
    74	    }
    75	  }
    76	
    77	  companion object {
    78	    private fun numeric(value: String): ByteArray? {
    79	      val valid =
    80	        if (':' in value) value.matches(Regex("[0-9a-fA-F:]+"))
    81	        else
    82	          value.split('.').let { p ->
    83	            p.size == 4 && p.all { it.matches(Regex("0|[1-9][0-9]{0,2}")) && it.toInt() <= 255 }
    84	          }
    85	      return if (valid) runCatching { InetAddress.getByName(value).address }.getOrNull() else null
    86	    }
    87	  }
    88	}

    90	/** 중복 헤더·다른 호출자의 키를 거부하며 인증 실패도 같은 응답으로 반환한다(POL-SEC-003). */

    93	  private val mapper: JsonMapper,
    94	) : OncePerRequestFilter() {

    98	
    99	  override fun shouldNotFilter(request: HttpServletRequest): Boolean {
   100	    val path = request.servletPath.ifEmpty { request.requestURI.removePrefix(request.contextPath) }
   101	    if (path.split('/').any { it == "." || it == ".." }) return false
   102	    return path == "/error" ||
   103	      (request.method in setOf("GET", "HEAD") &&
   104	        (path == "/actuator/health" || path.startsWith("/actuator/health/")))
   105	  }
   106	
   107	  override fun doFilterInternal(
   108	    request: HttpServletRequest,
   109	    response: HttpServletResponse,
   110	    chain: FilterChain,
   111	  ) {
   112	    val ids = request.getHeaders(SERVICE_HEADER).toList()
   113	    val values = request.getHeaders(HEADER).toList()
   114	    val caller = ids.singleOrNull()
   115	    val expected = keys[caller]
   116	    val reason =
   117	      when {
   118	        ids.size != 1 || values.size != 1 -> "HEADER_COUNT"
   119	        expected == null -> "UNKNOWN_CLIENT"
   120	        !MessageDigest.isEqual(expected, digest(values.single())) -> "KEY_MISMATCH"
   121	        networks.getValue(requireNotNull(caller)).let {
   122	          it.isNotEmpty() && it.none { network -> network.contains(request.remoteAddr) }
   123	        } -> "ADDRESS_NOT_ALLOWED"
   124	        else -> "ALLOWED"
   125	      }
   126	    log.info(
   127	      "event=internal.auth.completed caller={} reason={}",
   128	      if (expected == null) "unknown" else caller,
   129	      reason,
   130	    )
   131	    if (reason == "ALLOWED") {
   132	      request.setAttribute("internalCaller", requireNotNull(caller))
   133	      chain.doFilter(request, response)
   134	      return
   135	    }
   136	    val code = ErrorCode.INTERNAL_ACCESS_DENIED
   137	    request.setAttribute("errorCode", code.code)
   138	    val problem =

   145	          }

   149	    response.characterEncoding = "UTF-8"
   150	    response.writer.write(mapper.writeValueAsString(problem))
   151	  }
