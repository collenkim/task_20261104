     8	/** 원시 입력의 반복·형식을 먼저 검증한다. 생략만 기본값으로 처리한다(POL-API-003). */
     9	class MemberSummaryRequest
    10	private constructor(
    11	  val page: String?,
    12	  val size: String?,
    13	  val sort: String?,
    14	) {
    15	  fun toQuery(): MemberSummaryQuery {
    16	    val number = integer(page, 0, 0)
    17	    val effectiveSize =
    18	      integer(size, MemberPageRequest.DEFAULT_SIZE, 1).coerceAtMost(MemberPageRequest.MAX_SIZE)
    19	    if (number.toLong() * effectiveSize > MemberPageRequest.MAX_OFFSET) invalid()
    20	    val direction =
    21	      when (sort) {
    22	        null,
    23	        "name,asc" -> MemberNameSort.ASCENDING
    24	        "name,desc" -> MemberNameSort.DESCENDING
    25	        else -> invalid()
    26	      }
    27	    return MemberSummaryQuery(MemberPageRequest(number, effectiveSize), direction)
    28	  }
    29	
    30	  companion object {
    31	    private val DIGITS = Regex("[0-9]+")
    32	
    33	    /** 순서나 값의 동일 여부와 무관하게 각 파라미터를 한 번만 받는다. */
    34	    fun from(parameters: Map<String, List<String>>): MemberSummaryRequest {
    35	      for (name in listOf("page", "size", "sort")) {
    36	        if (parameters.containsKey(name) && parameters.getValue(name).size != 1) invalid()
    37	      }
    38	      return MemberSummaryRequest(
    39	        parameters["page"]?.single(),
    40	        parameters["size"]?.single(),
    41	        parameters["sort"]?.single(),
    42	      )
    43	    }
    44	
    45	    private fun integer(raw: String?, default: Int, minimum: Int): Int {
    46	      if (raw == null) return default
    47	      if (!DIGITS.matches(raw)) invalid()
    48	      val value = raw.toIntOrNull() ?: invalid()
    49	      if (value < minimum) invalid()
    50	      return value
    51	    }
    52	
    53	    private fun invalid(): Nothing = throw InvalidMemberQueryException()
    54	  }

    24	 * INTERNAL_ERROR로 바꾼다.
    25	 */
    26	@RestControllerAdvice
    27	class ProblemDetailsExceptionHandler : ResponseEntityExceptionHandler() {
    28	  private val log = LoggerFactory.getLogger(javaClass)
    29	
    30	  @ExceptionHandler(InvalidMemberQueryException::class)
    31	  fun handleInvalidMemberQuery(): ProblemDetail = memberProblem(ErrorCode.VALIDATION_FAILED)
    32	
    33	  @ExceptionHandler(MemberNotFoundException::class)
    34	  fun handleMemberNotFound(): ProblemDetail = memberProblem(ErrorCode.NOT_FOUND)
    35	
    36	  private fun memberProblem(code: ErrorCode): ProblemDetail {
    37	    SafeErrorLog.record(code)
    38	    return ProblemDetail.forStatusAndDetail(HttpStatusCode.valueOf(code.status), code.description)
    39	      .apply {
    40	        title = code.codeName
    41	        setProperty(CODE, code.code)
    42	      }
    43	      .let(::MemberProblemDetail)
    44	  }
    45	
    46	  override fun handleExceptionInternal(
    47	    ex: Exception,
    48	    body: Any?,
    49	    headers: HttpHeaders,
    50	    statusCode: HttpStatusCode,
    51	    request: WebRequest,
    52	  ): ResponseEntity<Any>? {
    53	    val response = super.handleExceptionInternal(ex, body, headers, statusCode, request)
    54	    val problem = response?.body as? ProblemDetail
    55	    if (problem != null && problem.properties?.containsKey(CODE) != true) {
    56	      val code =
    57	        when (ex) {
    58	          is MethodArgumentNotValidException,
    59	          is HandlerMethodValidationException -> ErrorCode.VALIDATION_FAILED
    60	          else -> ErrorCode.of(statusCode)
    61	        }

    63	      problem.setProperty(CODE, code.code)
    64	      problem.title = code.codeName
    65	      if (problem.detail == null) problem.detail = code.description
    66	    }
    67	    return response?.let {
    68	      ResponseEntity(
    69	        if (problem != null) MemberProblemDetail(problem) else it.body,
    70	        it.headers,

    73	    }

     9	internal object SafeErrorLog {

    13	    val request =
    14	      (RequestContextHolder.getRequestAttributes() as? ServletRequestAttributes)?.request
    15	    request?.setAttribute("errorCode", code.code)

    20	    log.error(
    21	      "event=exception.handled requestId={} stage=error.mapping code={} exception={} frames={}",
    22	      MDC.get("requestId") ?: "-",
    23	      code.code,
    24	      error?.javaClass?.simpleName ?: code.code,
    25	      frames,
    26	    )
    27	  }
