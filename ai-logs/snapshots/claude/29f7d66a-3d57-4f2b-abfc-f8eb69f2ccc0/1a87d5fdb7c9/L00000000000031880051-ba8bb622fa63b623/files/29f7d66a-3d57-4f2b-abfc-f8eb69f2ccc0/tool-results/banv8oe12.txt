     6	class ExponentialRetryBackoff(baseDelay: Duration, maxDelay: Duration) {
     7	  private val baseMillis: Long
     8	  private val maxMillis: Long
     9	
    10	  init {
    11	    require(validDelays(baseDelay, maxDelay)) { "재시도 지연 설정이 허용 범위를 벗어났습니다." }
    12	    baseMillis = baseDelay.toMillis()
    13	    maxMillis = maxDelay.toMillis()
    14	  }
    15	
    16	  /** 1부터 시작하는 재시도 번호. cap에 도달하면 반복을 끝내 큰 번호에도 overflow가 없다. */
    17	  fun upperBoundMillis(retryNumber: Int): Long {
    18	    require(retryNumber >= 1) { "재시도 번호는 1 이상이어야 합니다." }
    19	    var bound = baseMillis
    20	    var number = 1
    21	    while (number < retryNumber && bound < maxMillis) {
    22	      bound = if (bound > maxMillis / 2) maxMillis else bound * 2
    23	      number++
    24	    }
    25	    return bound
    26	  }
    27	
    28	  companion object {
    29	    private val MIN_DELAY = Duration.ofMillis(1)
    30	    private val MAX_DELAY = Duration.ofSeconds(60)

    32	    /** 정수 밀리초 1~60000, base <= cap. 큰 Duration을 밀리초로 바꾸기 전에 검증한다. */
    33	    fun validDelays(baseDelay: Duration, maxDelay: Duration): Boolean =
    34	      validDuration(baseDelay) && validDuration(maxDelay) && baseDelay <= maxDelay

    37	      value >= MIN_DELAY && value <= MAX_DELAY && value.nano % 1_000_000 == 0
    38	  }

    38	  ): ClientHttpResponse {
    39	    val oldAttempt = MDC.get("attempt")
    40	    try {
    41	      var retries = 0
    42	      while (true) {
    43	        stopIfInterrupted()
    44	        request.headers.set("X-Internal-Attempt", (retries + 1).toString())
    45	        MDC.put("attempt", (retries + 1).toString())
    46	        log.info("event=remote.attempt.started attempt={}", retries + 1)
    47	        try {
    48	          return execution.execute(request, body).also {
    49	            log.info(
    50	              "event=remote.attempt.completed attempt={} status={}",
    51	              retries + 1,
    52	              it.statusCode.value(),
    53	            )
    54	          }
    55	        } catch (e: IOException) {
    56	          log.error(
    57	            "event=remote.failed attempt={} exception={}",
    58	            retries + 1,
    59	            e.javaClass.simpleName,
    60	          )
    61	          stopIfInterrupted()
    62	          if (retries >= maxRetries || !isRetryable(request, e)) throw e
    63	          retries++
    64	          val upper = backoff.upperBoundMillis(retries)
    65	          val delay = jitter.sample(upper)
    66	          check(delay in 0..upper) { "재시도 난수 결과가 허용 범위를 벗어났습니다." }
    67	          log.info(
    68	            "event=remote.retry.scheduled nextAttempt={} delayMs={} reason={}",
    69	            retries + 1,
    70	            delay,
    71	            e.javaClass.simpleName,
    72	          )
    73	          try {
    74	            sleeper.sleep(delay)
    75	          } catch (_: InterruptedException) {
    76	            Thread.currentThread().interrupt()
    77	            throw interrupted()
    78	          }
    79	          stopIfInterrupted()
    80	        }
    81	      }
    82	    } finally {
    83	      if (oldAttempt == null) MDC.remove("attempt") else MDC.put("attempt", oldAttempt)
    84	    }
    85	  }
    86	
    87	  private fun stopIfInterrupted() {
    88	    if (Thread.currentThread().isInterrupted) throw interrupted()
    89	  }

    92	
    93	  private fun isRetryable(request: HttpRequest, e: IOException): Boolean =
    94	    request.method in RETRYABLE_METHODS &&
    95	      generateSequence<Throwable>(e) { it.cause }
    96	        .any {
    97	          it is ConnectException || it is HttpTimeoutException || it is SocketTimeoutException
    98	        }
